// ================== SETUP ==================
void setup() {
  Serial.begin(9600);



  // Button pins
  pinMode(btnMode, INPUT_PULLUP);
  pinMode(btnOK, INPUT_PULLUP);
  pinMode(btnCancel, INPUT_PULLUP);
  pinMode(btnUp, INPUT_PULLUP);
  pinMode(btnDown, INPUT_PULLUP);


  Serial.println("Testing buttons...");
  Serial.print("btnMode (33): ");
  Serial.println(digitalRead(btnMode));
  Serial.print("btnOK (34): ");
  Serial.println(digitalRead(btnOK));
  Serial.print("btnCancel (35): ");
  Serial.println(digitalRead(btnCancel));
  Serial.print("btnUp (36): ");
  Serial.println(digitalRead(btnUp));
  Serial.print("btnDown (37): ");
  Serial.println(digitalRead(btnDown));
  // Load settings dari EEPROM
  loadSettingsFromEEPROM();



  // SD
  Serial.println("Initializing SD card...");
  if (SD_CARD_DETECT_PIN != -1) {
    pinMode(SD_CARD_DETECT_PIN, INPUT_PULLUP);
  }
  tryInitSD(true);



  // RTC + LCD
  Wire.begin();
  rtc.begin();
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("Weather Station");
  lcd.setCursor(0, 1);
  lcd.print("Init...");



  // Rain gauge ISR
  pinMode(pin_interrupt, INPUT);
  attachInterrupt(digitalPinToInterrupt(pin_interrupt), hitung_curah_hujan, FALLING);



  // Serial
  windSerial.begin(9600);
  windSerial.setTimeout(20);
  mySerial.begin(9600);



  // RS485
  rs485Serial.begin(4800);
  rs485.begin(1, rs485Serial);
  rs485Serial.listen();



  // GPS
  Serial2.begin(GPSBaud);
  Serial.println(F("GPS on Serial2"));



  commSlotStartMs = millis();
  switchCommSlot(CommSlot::WIND);



  // Anemometer ISR
  pinMode(GPIO_pulse, INPUT_PULLUP);
  detachInterrupt(digitalPinToInterrupt(GPIO_pulse));
  attachInterrupt(digitalPinToInterrupt(GPIO_pulse), rpm_anemometer, RISING);



  // DS18B20
  ds18.begin();



  // pH
  ph.begin();



  // Pumps
  pinMode(POMPA_AIR_LAUT_PIN, OUTPUT);
  pinMode(POMPA_AIR_BILAS_PIN, OUTPUT);
  digitalWrite(POMPA_AIR_LAUT_PIN, HIGH);   // OFF
  digitalWrite(POMPA_AIR_BILAS_PIN, HIGH);  // OFF



  delay(400);
  lcd.clear();
  updateLCD();
}



// ================== MODE BUTTON DEBOUNCE ==================
unsigned long lastModeButtonPress = 0;
const unsigned long MODE_BUTTON_DEBOUNCE = 1000; // 1 detik cooldown



// ================== RESET FUNCTION ==================
void resetBackgroundProcesses() {
  // Reset comm slot
  commSlotStartMs = millis();
 
  // Reset LCD
  lcdForce = true;
  needsRedraw = true;
 
  // Reset editing state
  isEditing = false;
  editingField = 0;
  blinkState = false;
 
  Serial.println("Background processes reset");
}



// ================== MAIN LOOP (REVISED) ==================
void loop() {
  unsigned long currentMillis = millis();


  // ========================================================
  // PRIORITAS TERTINGGI: CEK TOMBOL MODE (KUNING) DULU
  // ========================================================
  if (readButton(btnMode)) {
    // Cek apakah sudah lewat debounce time (1 detik)
    if (currentMillis - lastModeButtonPress >= MODE_BUTTON_DEBOUNCE) {
      lastModeButtonPress = currentMillis;
     
      // Toggle mode
      if (currentMode == MODE_DISPLAY) {
        currentMode = SETTING;
        settingIndex = 0;
        Serial.println(">>> MASUK MODE SETTING <<<");
      } else {
        currentMode = MODE_DISPLAY;
        Serial.println(">>> KEMBALI KE MODE DISPLAY <<<");
      }
     
      // Reset semua proses background
      resetBackgroundProcesses();
     
      // Update LCD
      if (currentMode == MODE_DISPLAY) {
        updateLCD();
      } else {
        displaySetting(settingIndex);
      }
     
      // Jangan proses apapun lagi di loop ini
      return;
    } else {
      // Belum 1 detik sejak tekan terakhir
      Serial.println("Tombol Mode masih cooldown...");
      delay(100);
      return;
    }
  }


  // === MODE DISPLAY: Monitor sensor & control ===
  if (currentMode == MODE_DISPLAY) {


    // Toggle COMM slot
    if (millis() - commSlotStartMs >= SLOT_COMM_MS) {
      switchCommSlot(commSlot == CommSlot::WIND ? CommSlot::RS485 : CommSlot::WIND);
    }


    bool windDirChanged = false;


    // SD auto-retry
    if (!sdOk) {
      bool canTry = true;
      if (SD_CARD_DETECT_PIN != -1) {
        canTry = (digitalRead(SD_CARD_DETECT_PIN) == LOW);
      }
      if (canTry && (millis() - lastSdAttemptMs >= sdRetryIntervalMs)) {
        lastSdAttemptMs = millis();
        tryInitSD(true);
      }
    }


    // Baca arah angin (windSerial)
    if (commSlot == CommSlot::WIND) {
      windSerial.listen();
      if (windSerial.available()) {
        data = windSerial.readString();
        a = data.indexOf("*");
        b = data.indexOf("#");
        if (a >= 0 && b > a) {
          s_angin = data.substring(a + 1, b);
          String oldShort = arah_singkat;
          if (s_angin == "1") {
            arah_angin = "utara";
            arah_singkat = "U";
          } else if (s_angin == "2") {
            arah_angin = "timur laut";
            arah_singkat = "TL";
          } else if (s_angin == "3") {
            arah_angin = "timur";
            arah_singkat = "T";
          } else if (s_angin == "4") {
            arah_angin = "tenggara";
            arah_singkat = "TG";
          } else if (s_angin == "5") {
            arah_angin = "selatan";
            arah_singkat = "S";
          } else if (s_angin == "6") {
            arah_angin = "barat daya";
            arah_singkat = "BD";
          } else if (s_angin == "7") {
            arah_angin = "barat";
            arah_singkat = "B";
          } else if (s_angin == "8") {
            arah_angin = "barat laut";
            arah_singkat = "BL";
          } else {
            arah_angin = "unknown";
            arah_singkat = "--";
          }
          if (arah_singkat != oldShort) windDirChanged = true;
        }
      }
    }


    // Baca TH via RS485
    if (commSlot == CommSlot::RS485) {
      rs485Serial.listen();
      readTH_RS485();
    }


    // GPS feed
    processGPS();


    // Sample sensor
    sampleKecepatanInstan();


    // Suhu air
    float tempC_new = readDS18x20C();


    // TSS
    float vTurb_new = readVoltageAvg(TURB_PIN);
    float vTurb_V = vTurb_new;
    float tssNTU;
    if (vTurb_V >= V_NTU0) tssNTU = 0.0;
    else if (vTurb_V <= V_NTU3000) tssNTU = 3000.0;
    else if (vTurb_V > V_NTU1000) {
      tssNTU = ((V_NTU0 - vTurb_V) / (V_NTU0 - V_NTU1000)) * 1000.0;
    } else {
      tssNTU = 1000.0 + ((V_NTU1000 - vTurb_V) / (V_NTU1000 - V_NTU3000)) * 2000.0;
    }
    g_tss_NTU = tssNTU;
    g_vTurb = vTurb_V;
    debugTSS(vTurb_V, tssNTU);


    // TDS
    static unsigned long tdsSampleMs = millis();
    if (millis() - tdsSampleMs > 40) {
      tdsSampleMs = millis();
      tdsBuf[tdsIdx] = analogRead(TDS_PIN);
      tdsIdx = (tdsIdx + 1) % TDS_SCOUNT;
    }
    static unsigned long tdsCalcMs = millis();
    if (millis() - tdsCalcMs > 800) {
      tdsCalcMs = millis();
      int med = medianFilter(tdsBuf, TDS_SCOUNT);
      float vAvg = med * (VREF_TDS_MEAS / 1023.0f);
      float tempUsedC = (g_tempC <= -999) ? 25.0f : g_tempC;
      float comp = 1.0f + 0.02f * (tempUsedC - 25.0f);
      float vComp = vAvg / comp;
      float tdsRaw = (133.42f * vComp * vComp * vComp - 255.86f * vComp * vComp + 857.39f * vComp) * TDS_FACTOR;
      g_tdsPPM = tdsRaw * CAL_FACTOR;
      // EC derived from TDS: EC (mS/cm) = TDS (ppm) / 500
      g_ec_mScm = g_tdsPPM / 500.0f;
      g_tdsAvgVolt = vAvg;
      debugTDS(med, vAvg, vComp, tdsRaw, g_tdsPPM);
    }


    // pH
    long phSum = 0;
    const int PH_N = 10;
    for (int i = 0; i < PH_N; i++) {
      phSum += analogRead(PH_PIN);
      delay(2);
    }
    float phAdc = phSum / (float)PH_N;
    float vPH_mV = (phAdc / 1023.0f) * 5000.0f;


    if (isnan(g_tempC) || fabs(g_tempC - tempC_new) > 0.2f) {
      g_tempC = tempC_new;
    }
    float tempUsedC = (g_tempC <= -999) ? 25.0f : g_tempC;


    float pH_new = ph.readPH(vPH_mV, tempUsedC);
    g_pH = pH_new;
    debugPH(phAdc, vPH_mV, tempUsedC, pH_new);


    // DO
    long doSum = 0;
    const int DO_N = 20;
    for (int i = 0; i < DO_N; i++) {
      doSum += analogRead(DO_PIN);
      delay(2);
    }
    uint16_t doRaw = doSum / DO_N;
    float do_mV = doRaw * (VREF_MV / ADC_COUNTS);
    g_do_mV = do_mV;
    float doSat = DOsaturationAtTemp(tempUsedC);
    float frac = (do_mV - DO_VZERO_mV) / (DO_VAIR_mV - DO_VZERO_mV);
    if (frac < 0) frac = 0;
    if (frac > 1.2f) frac = 1.2f;
    g_do_mgL = frac * doSat;
    debugDO(doRaw, do_mV, tempUsedC, doSat, g_do_mgL);


    // Tombol untuk ganti slide MANUAL (Up/Down aja, Mode sudah di handle di atas)
    if (readButton(btnUp)) {
      currentSlide = (currentSlide + 1) % 6;
      lcdForce = true;
      delay(200);
    } else if (readButton(btnDown)) {
      currentSlide = (currentSlide - 1 + 6) % 6;
      lcdForce = true;
      delay(200);
    }


    // Per detik
    if (currentMillis - previousMillis >= interval) {
      previousMillis = currentMillis;
      bacaRTC();


      prosesPerintahSerialPompa();
      kontrolPompaTerjadwal();  // Kontrol pompa dari EEPROM


      if (menit != previousMenit) {
        float kecepatan_rata_rata = (count_speed > 0) ? (kecepatan_total / count_speed) : 0.0f;
        curah_hujan = jumlah_tip * milimeter_per_tip;


        // SD write
        String latStr = gpsEverFixed ? String(gpsLat, 6) : "searching";
        String lngStr = gpsEverFixed ? String(gpsLng, 6) : "searching";


        bool canMeasure = pompaLautOn;
        String tssField = canMeasure ? (String(g_tss_NTU, 1) + " NTU") : "tidak mengukur";
        String tdsField = canMeasure ? (String(g_tdsPPM, 0) + " ppm") : "tidak mengukur";
        String wtempField = canMeasure ? (String(g_tempC, 2) + " C") : "tidak mengukur";
        String phField = canMeasure ? (String(g_pH, 2)) : "tidak mengukur";
        String doField = canMeasure ? (String(g_do_mgL, 2) + " mg/L") : "tidak mengukur";
        String ecField = canMeasure ? (String(g_ec_mScm, 2) + " ms/cm") : "tidak mengukur";


        String line = String("timestamp: ") + jam + ":" + menit + ":" + detik
                      + " | wind_direction: " + arah_angin
                      + " | wind_speed: " + String(kecepatan_rata_rata, 2) + " km/h"
                      + " | rainfall: " + String(curah_hujan, 2) + " mm"
                      + " | TSS: " + tssField
                      + " | TDS: " + tdsField
                      + " | water temp: " + wtempField
                      + " | pH: " + phField
                      + " | DO: " + doField
                      + " | EC: " + ecField
                      + " | air temp: " + (isnan(g_airTempC) ? String("0") : String(g_airTempC, 1)) + " C"
                      + " | air humidity: " + (isnan(g_airRH) ? String("0") : String(g_airRH, 1)) + " %"
                      + " | pompaAirLaut: " + (pompaLautOn ? "ON" : "OFF")
                      + " | pompaBilas: " + (pompaBilasOn ? "ON" : "OFF")
                      + " | Latitude: " + latStr
                      + " | Longitude: " + lngStr;


        if (sdAppendLine(line)) {
          Serial.println("Data written to DataLog.txt");
        } else {
          Serial.println("SD not ready");
        }


        // Kirim ke ESP32
        bool canMeasure2 = pompaLautOn;
        String tssSend = canMeasure2 ? String(g_tss_NTU, 1) : "tidak mengukur";
        String tdsSend = canMeasure2 ? String(g_tdsPPM, 0) : "tidak mengukur";
        String wtempSend = canMeasure2 ? String(g_tempC, 2) : "tidak mengukur";
        String phSend = canMeasure2 ? String(g_pH, 2) : "tidak mengukur";
        String doSend = canMeasure2 ? String(g_do_mgL, 2) : "tidak mengukur";
        String ecSend = canMeasure2 ? String(g_ec_mScm, 2) : "tidak mengukur";


        String dataToSend = String("wind_direction: ") + arah_angin
                            + ", wind_speed: " + String(kecepatan_rata_rata, 2)
                            + ", rainfall: " + String(curah_hujan, 2)
                            + ", TSS: " + tssSend
                            + ", TDS: " + tdsSend
                            + ", water temp: " + wtempSend
                            + ", pH: " + phSend
                            + ", DO: " + doSend
                            + ", EC: " + ecSend
                            + ", pompaAirLaut: " + (pompaLautOn ? "ON" : "OFF")
                            + ", pompaBilas: " + (pompaBilasOn ? "ON" : "OFF")
                            + ", Latitude: " + latStr + ", Longitude: " + lngStr
                            + ", air temp: " + (isnan(g_airTempC) ? String("0") : String(g_airTempC, 1))
                            + ", air humidity: " + (isnan(g_airRH) ? String("0") : String(g_airRH, 1));


        mySerial.println(dataToSend);
        Serial.print("To ESP32 -> ");
        Serial.println(dataToSend);


        previousMenit = menit;
        noInterrupts();
        jumlah_tip = 0;
        interrupts();
        kecepatan_total = 0;
        count_speed = 0;
        avgSpeedForDisplay = 0.0f;
        lcdForce = true;
      }
    }


    // Event & refresh LCD
    noInterrupts();
    bool rgEvent = flag;
    flag = false;
    interrupts();


    if (rgEvent || windDirChanged || lcdForce) {
      if (millis() - prevLcdMillis >= lcdMinInterval) {
        updateLCD();
        lcdForce = false;
      }
    }


    // Kalibrasi pH
    {
      long sum = 0;
      const int N = 10;
      for (int i = 0; i < N; i++) {
        sum += analogRead(PH_PIN);
        delay(2);
      }
      float adc = sum / (float)N;
      float vPH_mV_now = (adc / 1023.0f) * 5000.0f;
      ph.calibration(vPH_mV_now, tempUsedC);
    }


  }
  // === MODE SETTING: Handle tombol setting ===
  else if (currentMode == SETTING) {
    handleSettingMode();
  }


  delay(50);
}



// ================== LOAD SETTINGS DARI EEPROM ==================
void loadSettingsFromEEPROM() {
  for (int i = 0; i < 2; i++) {
    for (int j = 0; j < 6; j++) {
      int addr = (i * 6 + j) * 2;
      settings[i].data[j] = EEPROM.read(addr) | (EEPROM.read(addr + 1) << 8);


      // Validasi
      if (settings[i].data[j] == -1 || settings[i].data[j] > 1000) {
        settings[i].data[j] = 0;
      }
    }
  }


  // Set default untuk repeat interval
  if (settings[IDX_POMPA_LAUT].data[2] == 0) settings[IDX_POMPA_LAUT].data[2] = 1;
  if (settings[IDX_POMPA_BILAS].data[2] == 0) settings[IDX_POMPA_BILAS].data[2] = 1;


  Serial.println("Settings loaded from EEPROM");
}



// ================== KONTROL POMPA TERJADWAL (DARI EEPROM) ==================
void kontrolPompaTerjadwal() {
  int m = menit.toInt();
  int h = jam.toInt();
  int d = tanggal.toInt();


  // ==== POMPA LAUT ====
  if (!overrideLaut) {
    int repHari = settings[IDX_POMPA_LAUT].data[2];
    int repJam = settings[IDX_POMPA_LAUT].data[3];
    int waktuJam = settings[IDX_POMPA_LAUT].data[4];
    int waktuMin = settings[IDX_POMPA_LAUT].data[5];
    int durMin = settings[IDX_POMPA_LAUT].data[0];
    int durSec = settings[IDX_POMPA_LAUT].data[1];
    unsigned long durasi = (durMin * 60000UL) + (durSec * 1000UL);


    bool shouldRun = false;


    // Cek repeat hari (modulo tanggal)
    if (d % repHari == 0) {
      if (repJam > 0) {
        // Nyala setiap X jam di menit waktuMin
        if (h % repJam == 0 && m == waktuMin) {
          shouldRun = true;
        }
      } else {
        // Nyala 1x sehari di waktuJam:waktuMin
        if (h == waktuJam && m == waktuMin) {
          shouldRun = true;
        }
      }
    }


    if (shouldRun && !pompaLautOn) {
      nyalakanPompaLaut();
    }


    // Matikan setelah durasi
    if (pompaLautOn && (millis() - pompaLautStartMs >= durasi)) {
      matikanPompaLaut();
    }
  }


  // ==== POMPA BILAS ====
  if (!overrideBilas) {
    int repHari = settings[IDX_POMPA_BILAS].data[2];
    int repJam = settings[IDX_POMPA_BILAS].data[3];
    int waktuJam = settings[IDX_POMPA_BILAS].data[4];
    int waktuMin = settings[IDX_POMPA_BILAS].data[5];
    int durMin = settings[IDX_POMPA_BILAS].data[0];
    int durSec = settings[IDX_POMPA_BILAS].data[1];
    unsigned long durasi = (durMin * 60000UL) + (durSec * 1000UL);


    bool shouldRun = false;


    // Cek repeat hari (modulo tanggal)
    if (d % repHari == 0) {
      if (repJam > 0) {
        // Nyala setiap X jam di menit waktuMin
        if (h % repJam == 0 && m == waktuMin) {
          shouldRun = true;
        }
      } else {
        // Nyala 1x sehari di waktuJam:waktuMin
        if (h == waktuJam && m == waktuMin) {
          shouldRun = true;
        }
      }
    }


    if (shouldRun && !pompaBilasOn) {
      nyalakanPompaBilas();
    }


    // Matikan setelah durasi
    if (pompaBilasOn && (millis() - pompaBilasStartMs >= durasi)) {
      matikanPompaBilas();
    }
  }
}

bool readButton(int pin) {

void handleSettingMode() {
  static int lastSettingIndex = -1;


  if (!isEditing) {
    if (needsRedraw || settingIndex != lastSettingIndex) {
      displaySetting(settingIndex);
      lastSettingIndex = settingIndex;
      needsRedraw = false;
    }


    if (readButton(btnUp)) {
      settingIndex = (settingIndex + 1) % TOTAL_SETTINGS;
      needsRedraw = true;
      delay(200);
    } else if (readButton(btnDown)) {
      settingIndex = (settingIndex - 1 + TOTAL_SETTINGS) % TOTAL_SETTINGS;
      needsRedraw = true;
      delay(200);
    } else if (readButton(btnOK)) {
      isEditing = true;
      editingField = 0;
      blinkState = true;
      lastBlinkUpdate = millis();
      displaySetting(settingIndex);
      Serial.println("Mulai edit setting: " + String(settingIndex));
      delay(200);
    }
    // HAPUS TOMBOL MODE DARI SINI - sudah di handle di loop() utama
   
  } else {
    // Mode editing - blink tiap 500ms
    unsigned long currentMillis = millis();
    if (currentMillis - lastBlinkUpdate >= 500) {
      lastBlinkUpdate = currentMillis;
      blinkState = !blinkState;
      displayWithBlink(settingIndex, editingField, blinkState);
    }


    if (readButton(btnOK)) {
      int maxField = getMaxField(settingIndex);
      if (editingField < maxField - 1) {
        // Skip field 4 (waktu jam) kalau repJam > 0
        if (editingField == 3 && (settingIndex == IDX_POMPA_LAUT || settingIndex == IDX_POMPA_BILAS)) {
          int repJam = settings[settingIndex].data[3];
          if (repJam > 0) {
            editingField = 5;  // Skip ke field 5
          } else {
            editingField++;
          }
        } else {
          editingField++;
        }
        blinkState = true;
        displayWithBlink(settingIndex, editingField, blinkState);
        delay(200);
      } else {
        saveSetting(settingIndex);
        isEditing = false;
        needsRedraw = true;
        displaySetting(settingIndex);
        Serial.println("Setting disimpan");
        delay(200);
      }
    } else if (readButton(btnCancel)) {
      isEditing = false;
      needsRedraw = true;
      displaySetting(settingIndex);
      Serial.println("Edit dibatalkan");
      delay(200);
    } else if (readButton(btnUp)) {
      handleValueChange(settingIndex, editingField, true);
      blinkState = true;
      displayWithBlink(settingIndex, editingField, blinkState);
      lastBlinkUpdate = millis();
      delay(150);
    } else if (readButton(btnDown)) {
      handleValueChange(settingIndex, editingField, false);
      blinkState = true;
      displayWithBlink(settingIndex, editingField, blinkState);
      lastBlinkUpdate = millis();
      delay(150);
    }
  }
}

void displayWithBlink(int index, int field, bool state) {
  displaySetting(index);


  if (!state) {
    if (index == IDX_POMPA_LAUT || index == IDX_POMPA_BILAS) {
      int repJam = settings[index].data[3];


      switch (field) {
        case 0:  // Durasi menit
          lcd.setCursor(8, 1);
          lcd.print("  ");
          break;
        case 1:  // Durasi detik
          lcd.setCursor(12, 1);
          lcd.print("  ");
          break;
        case 2:  // Repeat hari
          lcd.setCursor(8, 2);
          lcd.print("  ");
          break;
        case 3:  // Repeat jam
          lcd.setCursor(13, 2);
          lcd.print("  ");
          break;
        case 4:  // Waktu jam
          if (repJam == 0) {
            lcd.setCursor(7, 3);
            lcd.print("  ");
          }
          break;
        case 5:  // Waktu menit
          if (repJam > 0) {
            lcd.setCursor(7, 3);
            lcd.print("  ");
          } else {
            lcd.setCursor(10, 3);
            lcd.print("  ");
          }
          break;
      }
    }
  }
}



void handleValueChange(int index, int field, bool increment) {
  int delta = increment ? 1 : -1;


  if (index == IDX_POMPA_LAUT || index == IDX_POMPA_BILAS) {
    int repJam = settings[index].data[3];


    switch (field) {
      case 0:  // Durasi menit (0-59)
        settings[index].data[0] += delta;
        if (settings[index].data[0] > 59) settings[index].data[0] = 0;
        if (settings[index].data[0] < 0) settings[index].data[0] = 59;
        break;
      case 1:  // Durasi detik (0-59)
        settings[index].data[1] += delta;
        if (settings[index].data[1] > 59) settings[index].data[1] = 0;
        if (settings[index].data[1] < 0) settings[index].data[1] = 59;
        break;
      case 2:  // Repeat hari (1-30)
        settings[index].data[2] += delta;
        if (settings[index].data[2] > 30) settings[index].data[2] = 1;
        if (settings[index].data[2] < 1) settings[index].data[2] = 30;
        break;
      case 3:  // Repeat jam (0-24)
        settings[index].data[3] += delta;
        if (settings[index].data[3] > 24) settings[index].data[3] = 0;
        if (settings[index].data[3] < 0) settings[index].data[3] = 24;
        break;
      case 4:  // Waktu jam (0-23)
        if (repJam == 0) {
          settings[index].data[4] += delta;
          if (settings[index].data[4] > 23) settings[index].data[4] = 0;
          if (settings[index].data[4] < 0) settings[index].data[4] = 23;
        }
        break;
      case 5:  // Waktu menit (0-59)
        settings[index].data[5] += delta;
        if (settings[index].data[5] > 59) settings[index].data[5] = 0;
        if (settings[index].data[5] < 0) settings[index].data[5] = 59;
        break;
    }
  }
}



int getMaxField(int index) {
  if (index == IDX_POMPA_LAUT || index == IDX_POMPA_BILAS) {
    return 6;  // Field 0-5
  }
  return 1;
}



void saveSetting(int index) {
  for (int j = 0; j < 6; j++) {
    int addr = (index * 6 + j) * 2;
    int value = settings[index].data[j];
    EEPROM.update(addr, value & 0xFF);
    EEPROM.update(addr + 1, (value >> 8) & 0xFF);
  }
  Serial.println("Setting " + String(index) + " disimpan ke EEPROM");
}



String getSettingString(int index) {
  if (index == IDX_POMPA_LAUT) return "Pompa Laut";
  else if (index == IDX_POMPA_BILAS) return "Pompa Bilas";
  return "";
}

// ================== RTC READ ==================
void bacaRTC() {
  String t = rtc.getTimeStr();
  if (t.length() >= 8) {
    jam = t.substring(0, 2);
    menit = t.substring(3, 5);
    detik = t.substring(6, 8);
  } else {
    jam = "00";
    menit = "00";
    detik = "00";
  }


  String d = rtc.getDateStr();  // Format: DD.MM.YYYY
  if (d.length() >= 2) {
    tanggal = d.substring(0, 2);
  } else {
    tanggal = "01";
  }
}

  while (Serial2.available() > 0) {

  }

void rpm_anemometer() {
  unsigned long now = micros();
  if ((long)(now - lastPulseMicros) >= (long)debounceMicros) {
    pulseCount++;
    lastPulseMicros = now;
  }
}



void sampleKecepatanInstan() {
  if (millis() - lastSpeedSampleMs < speedSampleIntervalMs) return;
  unsigned long nowMs = millis();
  unsigned long dtMs = nowMs - lastSpeedSampleMs;
  lastSpeedSampleMs = nowMs;



  noInterrupts();
  unsigned long pulsesNow = pulseCount;
  interrupts();



  unsigned long dPulses = pulsesNow - prevPulseCount;
  prevPulseCount = pulsesNow;



  float dt = dtMs / 1000.0f;
  float rps = (PULSES_PER_ROT > 0) ? (dPulses / (dt * PULSES_PER_ROT)) : 0.0f;



  float v_ms = (-0.0181f * rps * rps) + (1.3859f * rps) + 1.4055f;
  if (v_ms <= 1.5f) v_ms = 0.0f;
  speed_instant_kmh = v_ms * 3.6f;



  kecepatan_total += speed_instant_kmh;
  count_speed++;
}

void prosesPerintahSerialPompa() {
  if (!Serial.available()) return;
  String input = Serial.readStringUntil('\n');
  input.trim();



  if (input == "cek") {
    cekStatusPompa();
    return;
  }



  if (input == "ON1") {
    overrideLaut = true;
    nyalakanPompaLaut();
    Serial.println("Pompa Air Laut Nyala (manual override ON)");
  } else if (input == "OFF1") {

  } else if (input == "OFF2") {
    matikanPompaBilas();
    overrideBilas = false;
    Serial.println("Pompa Bilas Mati (manual override OFF)");
  }
}

int medianFilter(int *src, int n) {
  for (int i = 0; i < n; i++) tdsBufTmp[i] = src[i];
  for (int j = 0; j < n - 1; j++)
    for (int i = 0; i < n - 1 - j; i++)
      if (tdsBufTmp[i] > tdsBufTmp[i + 1]) {
        int t = tdsBufTmp[i];
        tdsBufTmp[i] = tdsBufTmp[i + 1];
        tdsBufTmp[i + 1] = t;
      }
  if (n & 1) return tdsBufTmp[(n - 1) / 2];
  return (tdsBufTmp[n / 2] + tdsBufTmp[n / 2 - 1]) / 2;
}



float DOsaturationAtTemp(float tC) {
  struct P {
    float t;
    float d;
  };
  const P tab[] = { { 0, 14.6 }, { 10, 11.0 }, { 20, 9.1 }, { 25, 8.26 }, { 30, 7.6 } };
  if (tC <= tab[0].t) return tab[0].d;
  if (tC >= tab[4].t) return tab[4].d;
  for (int i = 0; i < 4; i++) {
    if (tC <= tab[i + 1].t) {
      float t1 = tab[i].t, d1 = tab[i].d;
      float t2 = tab[i + 1].t, d2 = tab[i + 1].d;
      return d1 + (d2 - d1) * (tC - t1) / (t2 - t1);
    }
  }
  return 8.26f;
}
